cmake_minimum_required(VERSION 3.20)
project(shinki_leaf LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(SHINKI "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(UPSTREAM "" CACHE PATH "Patched whitealbum2-Android source tree")
if(NOT EXISTS "${UPSTREAM}/engine/scene_voices.cpp")
    message(FATAL_ERROR "Pass the patched engine checkout with -DUPSTREAM=...")
endif()
option(LEAF_BUNDLED_SDL "Build SDL from the pinned engine checkout" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CODECS REQUIRED IMPORTED_TARGET libavformat libavcodec libswscale libswresample libavutil)
if(LEAF_BUNDLED_SDL)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_CCACHE OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC OFF CACHE BOOL "" FORCE)
    set(SDL_SHARED ON CACHE BOOL "" FORCE)
    add_subdirectory("${UPSTREAM}/tools/SDL-release-2.32.10" sdl EXCLUDE_FROM_ALL)
else()
    pkg_check_modules(SYSTEM_SDL REQUIRED IMPORTED_TARGET sdl2)
    add_library(SDL2 INTERFACE)
    target_link_libraries(SDL2 INTERFACE PkgConfig::SYSTEM_SDL)
endif()
find_package(Threads REQUIRED)
add_subdirectory("${UPSTREAM}/engine" engine EXCLUDE_FROM_ALL)
target_link_libraries(wa2_engine PUBLIC Threads::Threads)
add_library(wa2_movie STATIC "${UPSTREAM}/engine/movie_reader.cpp" "${UPSTREAM}/engine/movie_player.cpp")
target_link_libraries(wa2_movie PUBLIC wa2_engine PkgConfig::CODECS)
add_library(wa2_wave_codec STATIC "${UPSTREAM}/engine/wave_codec.cpp")
target_link_libraries(wa2_wave_codec PUBLIC wa2_engine PkgConfig::CODECS)
add_subdirectory("${UPSTREAM}/game" game EXCLUDE_FROM_ALL)
include("${UPSTREAM}/platform/native_services.cmake")
include("${UPSTREAM}/platform/sdl_canvas.cmake")
add_library(wa2_trial SHARED adapter.cpp
    "${UPSTREAM}/platform/sdl_presentation.cpp"
    "${UPSTREAM}/platform/sdl_application.cpp"
    "${UPSTREAM}/platform/sdl_audio.cpp")
target_include_directories(wa2_trial PRIVATE "${SHINKI}/Source/FamilyApi/shinki-emu-abi/include")
target_link_libraries(wa2_trial PRIVATE wa2_native_desktop_resources wa2_native_application
    wa2_native_audio_binding wa2_game_scene wa2_canvas wa2_movie wa2_wave_codec)
set_target_properties(wa2_trial PROPERTIES OUTPUT_NAME shinki_emu_wa2_trial)
if(MINGW)
    target_link_options(wa2_trial PRIVATE -static-libgcc -static-libstdc++)
    set_target_properties(wa2_trial PROPERTIES PREFIX "")
endif()
if(UNIX AND NOT APPLE)
    set_target_properties(wa2_trial PROPERTIES BUILD_RPATH_USE_ORIGIN ON)
endif()
